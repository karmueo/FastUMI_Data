cmake_minimum_required(VERSION 3.18)
# Ubuntu 22.04 的 JsonCpp config 仍声明旧 policy；兼容 CMake 4 的检查。
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(NOT CMAKE_CUDA_COMPILER AND NOT DEFINED ENV{CUDACXX})
  find_program(CMAKE_CUDA_COMPILER nvcc HINTS /usr/local/cuda/bin)
endif()
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 87)
endif()
project(dp_infer_tensorrt LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 87)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(fastumi_interfaces REQUIRED)
find_package(pluginlib REQUIRED)
find_package(urdf REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(orocos_kdl REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(OpenSSL REQUIRED)
find_package(jsoncpp REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
find_library(TENSORRT_LIBRARY nvinfer REQUIRED)

add_library(dp_infer_tensorrt_core SHARED src/core.cpp src/visual_guard.cpp)
target_include_directories(dp_infer_tensorrt_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(dp_infer_tensorrt_core PUBLIC Eigen3::Eigen jsoncpp_lib ${OpenCV_LIBS}
  OpenSSL::Crypto ${orocos_kdl_LIBRARIES})
ament_target_dependencies(dp_infer_tensorrt_core PUBLIC urdf kdl_parser)

add_library(dp_infer_tensorrt_runtime SHARED src/runtime.cpp src/ddim.cu)
target_include_directories(dp_infer_tensorrt_runtime PRIVATE ${TENSORRT_INCLUDE_DIR})
target_link_libraries(dp_infer_tensorrt_runtime PUBLIC dp_infer_tensorrt_core PRIVATE CUDA::cudart ${TENSORRT_LIBRARY})
target_compile_options(dp_infer_tensorrt_runtime PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)

add_library(dp_infer_tensorrt_ros SHARED src/node.cpp)
target_link_libraries(dp_infer_tensorrt_ros PUBLIC dp_infer_tensorrt_runtime)
ament_target_dependencies(dp_infer_tensorrt_ros PUBLIC rclcpp rcl_interfaces fastumi_interfaces
  sensor_msgs std_msgs std_srvs geometry_msgs builtin_interfaces pluginlib)
add_executable(dp_infer_tensorrt_node src/main.cpp)
target_link_libraries(dp_infer_tensorrt_node dp_infer_tensorrt_ros)
add_executable(dp_infer_tensorrt_runner src/runner.cpp)
target_link_libraries(dp_infer_tensorrt_runner dp_infer_tensorrt_runtime)

foreach(target dp_infer_tensorrt_core dp_infer_tensorrt_runtime dp_infer_tensorrt_ros dp_infer_tensorrt_node dp_infer_tensorrt_runner)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endforeach()

install(TARGETS dp_infer_tensorrt_core dp_infer_tensorrt_runtime dp_infer_tensorrt_ros
  EXPORT dp_infer_tensorrt_targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS dp_infer_tensorrt_node dp_infer_tensorrt_runner DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core dp_infer_tensorrt_core)
  target_compile_definitions(test_core PRIVATE TEST_URDF_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../fastumi_rm75/assets/rm_75_kinematic.urdf")
  ament_add_gtest(test_node test/test_node.cpp ENV ROS_DOMAIN_ID=179 ROS_LOCALHOST_ONLY=1 TIMEOUT 90)
  target_link_libraries(test_node dp_infer_tensorrt_ros)
  ament_target_dependencies(test_node rosgraph_msgs)
  target_compile_definitions(test_node PRIVATE TEST_URDF_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../fastumi_rm75/assets/rm_75_kinematic.urdf")
  add_library(test_postprocessors SHARED test/test_postprocessors.cpp)
  target_link_libraries(test_postprocessors dp_infer_tensorrt_core)
  ament_target_dependencies(test_postprocessors pluginlib)
  pluginlib_export_plugin_description_file(dp_infer_tensorrt test/plugins.xml)
  install(TARGETS test_postprocessors LIBRARY DESTINATION lib)
endif()

ament_export_targets(dp_infer_tensorrt_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 jsoncpp OpenCV OpenSSL urdf kdl_parser orocos_kdl
  rclcpp rcl_interfaces fastumi_interfaces sensor_msgs std_msgs std_srvs geometry_msgs builtin_interfaces pluginlib)
ament_package(CONFIG_EXTRAS cmake/jsoncpp_compat.cmake)
