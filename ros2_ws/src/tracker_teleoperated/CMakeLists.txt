# 构建遥操 RViz 面板，并安装绑定 NumPy 2 解释器的 Python 节点。
cmake_minimum_required(VERSION 3.19)
project(tracker_teleoperated)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rviz_common REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(fastumi_interfaces REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
ament_python_install_package(${PROJECT_NAME})
foreach(entry tracker_teleop_node tracker_component_manager)
  configure_file(scripts/${entry}.in ${CMAKE_CURRENT_BINARY_DIR}/${entry} @ONLY)
  file(CHMOD ${CMAKE_CURRENT_BINARY_DIR}/${entry}
    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${entry} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_library(teleop_panel SHARED
  src/teleop_panel.cpp src/recordings_widget.cpp src/video_source_widget.cpp
  include/tracker_teleoperated/teleop_panel.hpp
  include/tracker_teleoperated/recordings_widget.hpp
  include/tracker_teleoperated/video_source_widget.hpp)
target_compile_features(teleop_panel PUBLIC cxx_std_17)
target_include_directories(teleop_panel PUBLIC include)
ament_target_dependencies(teleop_panel pluginlib rclcpp rcl_interfaces rviz_common std_msgs std_srvs diagnostic_msgs fastumi_interfaces)
target_link_libraries(teleop_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS teleop_panel DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES plugin_description.xml README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_pytest_test(test_python test TIMEOUT 120)
  ament_add_gtest(test_panel test/test_panel.cpp)
  target_include_directories(test_panel PRIVATE include)
  target_link_libraries(test_panel teleop_panel Qt5::Test Qt5::Widgets)
  ament_target_dependencies(test_panel rclcpp rviz_common std_msgs std_srvs diagnostic_msgs fastumi_interfaces)
  set_tests_properties(test_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  ament_add_gtest(test_recordings_widget test/test_recordings_widget.cpp)
  target_include_directories(test_recordings_widget PRIVATE include)
  target_link_libraries(test_recordings_widget teleop_panel Qt5::Test Qt5::Widgets)
  ament_target_dependencies(test_recordings_widget rclcpp fastumi_interfaces)
  set_tests_properties(test_recordings_widget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  ament_add_gtest(test_panel_recovery test/test_panel_recovery.cpp)
  target_include_directories(test_panel_recovery PRIVATE include)
  target_link_libraries(test_panel_recovery teleop_panel Qt5::Test Qt5::Widgets)
  ament_target_dependencies(test_panel_recovery rclcpp rviz_common std_msgs std_srvs diagnostic_msgs fastumi_interfaces)
  set_tests_properties(test_panel_recovery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
ament_package()
